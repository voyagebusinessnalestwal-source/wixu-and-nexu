<script>

function envoyer(){

let texte=document.getElementById("question").value;

if(texte=="") return;

let chat=document.getElementById("chat");

chat.innerHTML+="<div class='message user'>👦 "+texte+"</div>";

let t=texte.toLowerCase();

let reponse="Je ne connais pas encore cette réponse. Demande à ton professeur si besoin 😊";

if(t.includes("bonjour"))
reponse="Bonjour ! Comment puis-je t'aider ?";

else if(t.includes("2+2") || t.includes("2 + 2"))
reponse="2 + 2 = 4.";

else if(t.includes("5x5") || t.includes("5 x 5"))
reponse="5 × 5 = 25.";

else if(t.includes("capitale de la france"))
reponse="La capitale de la France est Paris.";

else if(t.includes("verbe"))
reponse="Un verbe est un mot qui indique une action ou un état.";

else if(t.includes("nom"))
reponse="Un nom sert à désigner une personne, un animal, un objet ou un lieu.";

else if(t.includes("adjectif"))
reponse="Un adjectif donne une précision sur un nom.";

else if(t.includes("merci"))
reponse="Avec plaisir ! Bon courage pour tes devoirs 📚";

chat.innerHTML+="<div class='message bot'>🤖 "+reponse+"</div>";

chat.scrollTop=chat.scrollHeight;

document.getElementById("question").value="";

}

document.getElementById("question").addEventListener("keypress",function(e){
if(e.key==="Enter"){
envoyer();
}
});

</script>
